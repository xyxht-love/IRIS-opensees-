EVENTS = [
  {"network": "II", "station": "LVZ", "time": "1993-11-27T10:45:29", "magnitude": 4.2},
  {"network": "II", "station": "LVZ", "time": "1995-08-31T13:03:27", "magnitude": 4.0},
  {"network": "IU", "station": "KEV", "time": "1993-09-03T11:21:20", "magnitude": 4.8},
  # ... 其余事件与之前完全一致（此处省略，实际使用时需补全）
  # 请务必复制完整的 EVENTS 列表（与 fetch.txt 相同）

# ============================================================
# 用户参数
# ============================================================
ACCEL_IN_G = False
EPS = 1.0e-12
M_ADD = 3000.0

F_H = 4.885353e+02
F_V = 8.001221e+01

D = 0.35
t_wall = 0.035
L = 4.5
rho = 2700.0
A_cross = np.pi * ((D/2)**2 - (D/2 - t_wall)**2)
BEAM_MASS = rho * A_cross * L

ACCEL_FILES = {
    'X': "MH1.txt",
    'Y': "MH2.txt",
    'Z': "combine.txt"
}

# ============================================================
# 自动下载（仍使用当前目录，但 fetch.txt 已输出到子目录）
# ============================================================
def ensure_seismic_files(event_index):
    # 检查 events/event_<index>/ 下的文件
    base_dir = f"events/event_{event_index}"
    missing = []
    for f in ACCEL_FILES.values():
        if not os.path.exists(os.path.join(base_dir, f)):
            missing.append(f)
    if missing:
        print(f"地震动文件缺失: {missing}，正在调用 fetch.txt 下载事件 {event_index} ...")
        try:
            subprocess.check_call([sys.executable, "fetch.txt", "--index", str(event_index)])
        except subprocess.CalledProcessError as e:
            raise RuntimeError(f"下载失败: {e}")
        except FileNotFoundError:
            raise RuntimeError("未找到 fetch.txt，请确保该脚本位于当前目录。")
        print("下载完成。")
    else:
        print("所有地震动文件已存在。")

# ============================================================
# 读取与计算函数
# ============================================================
def read_all_nodes(filepath):
    if not os.path.exists(filepath):
        raise FileNotFoundError(f"文件不存在: {filepath}")
    data = np.loadtxt(filepath)
    if data.ndim == 1:
        data = data.reshape(1, -1)
    t = data[:, 0]
    cols = data.shape[1] - 1
    nnodes = cols // 3
    u = data[:, 1:].reshape(-1, nnodes, 3)
    return t, u

def read_accel_file(fpath, dt_ref=0.01):
    data = np.loadtxt(fpath)
    if data.ndim == 1:
        data = data.reshape(-1, 1)
    if data.shape[1] == 1:
        n = data.shape[0]
        t = np.arange(n) * dt_ref
        accel = data[:, 0]
    else:
        t = data[:, 0]
        accel = data[:, 1]
    if ACCEL_IN_G:
        accel = accel * 9.80665
    return t, accel

def get_node_masses(n_mass_nodes, beam_mass, add_mass):
    if n_mass_nodes <= 1:
        return np.array([add_mass + beam_mass])
    m_internal = beam_mass / n_mass_nodes
    masses = np.full(n_mass_nodes, m_internal)
    masses[-1] = add_mass + 0.5 * m_internal
    return masses

def compute_edps(folder_path, case_id, event_info):
    disp_path = os.path.join(folder_path, "disp_all.out")
    vel_path  = os.path.join(folder_path, "vel_all.out")

    t, disp = read_all_nodes(disp_path)
    t_vel, vel = read_all_nodes(vel_path)
    if not np.allclose(t, t_vel):
        raise ValueError("位移与速度时间轴不一致！")

    dt = np.mean(np.diff(t))
    nt = len(t)
    nnodes = disp.shape[1]
    print(f"   读取: {nt} 步, dt = {dt:.6f} s, 质量节点数 = {nnodes}")

    # 读取加速度文件（从 events/event_<index>/ 目录）
    base_dir = f"events/event_{event_info['index']}"
    t_x, ax = read_accel_file(os.path.join(base_dir, ACCEL_FILES['X']), dt)
    t_y, ay = read_accel_file(os.path.join(base_dir, ACCEL_FILES['Y']), dt)
    t_z, az = read_accel_file(os.path.join(base_dir, ACCEL_FILES['Z']), dt)

    # 应用缩放
    ax = ax / F_H
    ay = ay / F_H
    az = az / F_V

    ag_x = np.interp(t, t_x, ax)
    ag_y = np.interp(t, t_y, ay)
    ag_z = np.interp(t, t_z, az)

    # 质量
    masses = get_node_masses(nnodes, BEAM_MASS, M_ADD)
    total_mass = np.sum(masses)

    # 顶部节点
    top_idx = -1
    ux_top = disp[:, top_idx, 0]
    uy_top = disp[:, top_idx, 1]
    uz_top = disp[:, top_idx, 2]
    vx_top = vel[:, top_idx, 0]
    vy_top = vel[:, top_idx, 1]
    vz_top = vel[:, top_idx, 2]
    disp_3d_top = np.sqrt(ux_top**2 + uy_top**2 + uz_top**2)

    # EDP 1
    D_cum_3D = np.trapz(disp_3d_top, t)

    # EDP 2
    abs_power_energy = np.zeros(nt)
    for i in range(nnodes):
        vx = vel[:, i, 0]
        vy = vel[:, i, 1]
        vz = vel[:, i, 2]
        abs_power_energy += masses[i] * (
            np.abs(ag_x * vx) +
            np.abs(ag_y * vy) +
            np.abs(ag_z * vz)
        )
    E_in = np.trapz(abs_power_energy, t)

    # EDP 3
    abs_power = np.zeros(nt)
    for i in range(nnodes):
        vx = vel[:, i, 0]
        vy = vel[:, i, 1]
        vz = vel[:, i, 2]
        abs_power += masses[i] * (
            np.abs(ag_x * vx) +
            np.abs(ag_y * vy) +
            np.abs(ag_z * vz)
        )
    cum_weight = np.cumsum(abs_power * dt)
    cum_weight_norm = cum_weight / cum_weight[-1]
    idx_5 = np.searchsorted(cum_weight_norm, 0.05)
    idx_95 = np.searchsorted(cum_weight_norm, 0.95)
    if idx_5 == idx_95:
        idx_5, idx_95 = 0, len(t) - 1
    D_w_peak = np.max(disp_3d_top[idx_5:idx_95+1])
    t5, t95 = t[idx_5], t[idx_95]

    # EDP 4
    D_3D_max = np.max(disp_3d_top)
    idx_max_3d = np.argmax(disp_3d_top)
    t_max_3d = t[idx_max_3d]

    # EDP 5
    int_vz2 = 0.0
    int_vh2 = 0.0
    for i in range(nnodes):
        vz = vel[:, i, 2]
        vx = vel[:, i, 0]
        vy = vel[:, i, 1]
        int_vz2 += masses[i] * np.trapz(vz**2, t)
        int_vh2 += masses[i] * np.trapz(vx**2 + vy**2, t)
    R_VH = int_vz2 / (int_vh2 + EPS)

    # 辅助
    max_uz = np.max(np.abs(uz_top))
    disp_h_top = np.sqrt(ux_top**2 + uy_top**2)
    max_h = np.max(disp_h_top)
    idx_max_uz = np.argmax(np.abs(uz_top))
    t_max_uz = t[idx_max_uz]

    return {
        "Case": case_id,
        "Folder": os.path.basename(folder_path),
        "event_index": event_info.get("index", None),
        "network": event_info.get("network", ""),
        "station": event_info.get("station", ""),
        "event_time": event_info.get("time", ""),
        "magnitude": event_info.get("magnitude", np.nan),
        "D_cum_3D (m·s)": D_cum_3D,
        "E_in (J)": E_in,
        "D_w_peak (m)": D_w_peak,
        "D_3D_max (m)": D_3D_max,
        "R_VH (-)": R_VH,
        "t5 (s)": t5,
        "t95 (s)": t95,
        "t@D3Dmax (s)": t_max_3d,
        "t@maxUz (s)": t_max_uz,
        "max|Uz| (m)": max_uz,
        "max|Uh| (m)": max_h,
        "总质量(动力) (kg)": total_mass,
        "梁质量 (kg)": BEAM_MASS,
        "附加质量 (kg)": M_ADD,
        "步长 dt (s)": dt,
    }

# 主程序
# ============================================================
def find_folders(event_index):
    # 查找 outputs/event_<event_index>/mesh_*/ 目录
    pattern = f"outputs/event_{event_index}/mesh_*"
    folders = glob.glob(pattern)
    # 过滤出目录
    folders = [f for f in folders if os.path.isdir(f)]
    # 按网格数排序
    folders.sort(key=lambda x: int(re.search(r'mesh_(\d+)', x).group(1)))
    return folders

def extract_case_id(folder_name):
    match = re.search(r'mesh_(\d+)', folder_name)
    if match:
        return f"mesh_{match.group(1)}"
    else:
        return os.path.basename(folder_name)

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="OpenSees EDP 后处理")
    parser.add_argument("--event-index", type=int, default=None,
                        help="指定事件索引，若未提供则读取环境变量 SEISMIC_EVENT_INDEX，否则默认为1")
    args = parser.parse_args()

    if args.event_index is not None:
        event_index = args.event_index
    else:
        event_index = int(os.environ.get("SEISMIC_EVENT_INDEX", "1"))

    if event_index < 1 or event_index > len(EVENTS):
        raise ValueError(f"事件索引 {event_index} 超出范围 (1~{len(EVENTS)})")

    event_info = EVENTS[event_index - 1].copy()
    event_info["index"] = event_index

    print("=" * 70)
    print("  OpenSees EDP 后处理 (适配事件子目录)")
    print(f"  当前事件: {event_info['network']}.{event_info['station']} (Index {event_index})")
    print("=" * 70)

    ensure_seismic_files(event_index)

    folders = find_folders(event_index)
    if not folders:
        print(f"❌ 未找到任何 outputs/event_{event_index}/mesh_* 文件夹！")
        print("   请确认 OpenSees 分析已成功完成。")
        sys.exit(1)

    print(f"找到 {len(folders)} 个文件夹：")
    for f in folders:
        print(f"  {f}")

    results = []
    for folder in folders:
        try:
            case_id = extract_case_id(os.path.basename(folder))
            print(f"\n处理 {folder} (Case={case_id}) ...")
            res = compute_edps(folder, case_id, event_info)
            results.append(res)
            print(f"\n  D_cum_3D = {res['D_cum_3D (m·s)']:.6e} m·s")
            print(f"  E_in     = {res['E_in (J)']:.6e} J")
            print(f"  D_w_peak = {res['D_w_peak (m)']:.6e} m")
            print(f"  D_3D_max = {res['D_3D_max (m)']:.6e} m")
            print(f"  R_VH     = {res['R_VH (-)']:.6e}")
        except Exception as e:
            print(f"  ❌ 处理失败: {e}")

    if not results:
        sys.exit(1)

    df = pd.DataFrame(results)
    csv_path = f"EDP_event_{event_index}_mesh_convergence.csv"
    df.to_csv(csv_path, index=False, float_format="%.6e")

    txt_path = f"EDP_event_{event_index}_mesh_convergence.txt"
    with open(txt_path, "w") as f:
        f.write(f"EDP Results for Event {event_index}\n")
        f.write("=" * 70 + "\n")
        f.write(df.to_string(index=False))
        f.write("\n" + "=" * 70 + "\n")
        f.write(f"事件: {event_info['network']}.{event_info['station']} (Index {event_index}, M{event_info['magnitude']})\n")
        f.write(f"缩放因子: 水平 F_H = {F_H}, 竖向 F_V = {F_V}\n")

    print(f"\n✅ 结果已保存至:")
    print(f"   - {csv_path}")
    print(f"   - {txt_path}")
