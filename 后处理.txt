# -*- coding: utf-8 -*-
"""
OpenSees EDP 后处理 (最终论文版 v7 - 从 model_info.txt 读取事件信息)
- 自动从 model_info.txt 读取事件信息
- 若缺失则回退到环境变量/默认值
- 其余 EDP 定义不变
"""

import os
import sys
import glob
import re
import numpy as np
import pandas as pd

# ============================================================
# 用户可调参数
# ============================================================
ACCEL_IN_G = False
EPS = 1.0e-12
M_ADD = 3000.0

D = 0.35
t_wall = 0.035
L = 4.5
rho = 2700.0
A_cross = np.pi * ((D/2)**2 - (D/2 - t_wall)**2)
BEAM_MASS = rho * A_cross * L

ACCEL_FILES = {
    'X': "MH1.txt",
    'Y': "MH2.txt",
    'Z': "combine.txt"
}

# ============================================================
# 读取函数
# ============================================================
def read_all_nodes(filepath):
    if not os.path.exists(filepath):
        raise FileNotFoundError(f"文件不存在: {filepath}")
    data = np.loadtxt(filepath)
    if data.ndim == 1:
        data = data.reshape(1, -1)
    t = data[:, 0]
    cols = data.shape[1] - 1
    nnodes = cols // 3
    u = data[:, 1:].reshape(-1, nnodes, 3)
    return t, u

def read_accel_file(fpath, dt_ref=0.01):
    data = np.loadtxt(fpath)
    if data.ndim == 1:
        data = data.reshape(-1, 1)
    if data.shape[1] == 1:
        n = data.shape[0]
        t = np.arange(n) * dt_ref
        accel = data[:, 0]
    else:
        t = data[:, 0]
        accel = data[:, 1]
    if ACCEL_IN_G:
        accel = accel * 9.80665
    return t, accel

def get_node_masses(n_mass_nodes, beam_mass, add_mass):
    if n_mass_nodes <= 1:
        return np.array([add_mass + beam_mass])
    m_internal = beam_mass / n_mass_nodes
    masses = np.full(n_mass_nodes, m_internal)
    masses[-1] = add_mass + 0.5 * m_internal
    return masses

def read_model_info(folder_path):
    """从 model_info.txt 读取 numEle, M_add, BeamMass 及事件信息"""
    info_path = os.path.join(folder_path, "model_info.txt")
    if not os.path.exists(info_path):
        raise FileNotFoundError(f"未找到 {info_path}")
    info = {}
    with open(info_path, 'r') as f:
        for line in f:
            parts = line.strip().split()
            if len(parts) >= 2:
                key = parts[0]
                value = " ".join(parts[1:])
                info[key] = value
    # 转换数值
    for key in ['numEle', 'M_add', 'BeamMass']:
        if key in info:
            try:
                info[key] = float(info[key])
            except:
                pass
    return info

# ============================================================
# EDP 计算核心（不变）
# ============================================================
def compute_edps(folder_path, case_id, event_info):
    # 读取 model_info 以获取 numEle 等（若有）
    try:
        model_info = read_model_info(folder_path)
        numEle_from_info = int(model_info.get('numEle', 10))
        M_add_from_info = float(model_info.get('M_add', M_ADD))
        beam_mass_from_info = float(model_info.get('BeamMass', BEAM_MASS))
    except:
        numEle_from_info = 10
        M_add_from_info = M_ADD
        beam_mass_from_info = BEAM_MASS

    disp_path = os.path.join(folder_path, "disp_all.out")
    vel_path  = os.path.join(folder_path, "vel_all.out")

    t, disp = read_all_nodes(disp_path)
    t_vel, vel = read_all_nodes(vel_path)
    if not np.allclose(t, t_vel):
        raise ValueError("位移与速度时间轴不一致！")

    dt = np.mean(np.diff(t))
    nt = len(t)
    nnodes = disp.shape[1]
    print(f"   读取: {nt} 步, dt = {dt:.6f} s, 质量节点数 = {nnodes}")

    # 诊断（保留）
    print("\n  [文件结构诊断]")
    raw_data = np.loadtxt(disp_path)
    total_cols = raw_data.shape[1]
    expected_cols = 1 + nnodes * 3
    print(f"  原始文件总列数: {total_cols}, 期望: {expected_cols}")
    if total_cols != expected_cols:
        print(f"  ⚠️ 警告：列数不匹配！")
    else:
        print(f"  ✅ 列数匹配")

    node2_disp = np.sqrt(disp[:, 0, 0]**2 + disp[:, 0, 1]**2 + disp[:, 0, 2]**2)
    top_disp = np.sqrt(disp[:, -1, 0]**2 + disp[:, -1, 1]**2 + disp[:, -1, 2]**2)
    print(f"  节点2 峰值位移: {np.max(node2_disp):.6e} m")
    print(f"  顶部节点 峰值位移: {np.max(top_disp):.6e} m")

    # 质量分配
    masses = get_node_masses(nnodes, beam_mass_from_info, M_add_from_info)
    total_mass = np.sum(masses)
    print(f"   节点质量总和 = {total_mass:.2f} kg (梁 {beam_mass_from_info:.2f} + 附加 {M_add_from_info:.2f})")

    # 读取地震波
    t_x, ax = read_accel_file(ACCEL_FILES['X'], dt)
    t_y, ay = read_accel_file(ACCEL_FILES['Y'], dt)
    t_z, az = read_accel_file(ACCEL_FILES['Z'], dt)
    ag_x = np.interp(t, t_x, ax)
    ag_y = np.interp(t, t_y, ay)
    ag_z = np.interp(t, t_z, az)

    # 顶部节点
    top_idx = -1
    ux_top = disp[:, top_idx, 0]
    uy_top = disp[:, top_idx, 1]
    uz_top = disp[:, top_idx, 2]
    vx_top = vel[:, top_idx, 0]
    vy_top = vel[:, top_idx, 1]
    vz_top = vel[:, top_idx, 2]
    disp_3d_top = np.sqrt(ux_top**2 + uy_top**2 + uz_top**2)

    # EDP 1
    D_cum_3D = np.trapz(disp_3d_top, t)

    # EDP 2
    abs_power_energy = np.zeros(nt)
    for i in range(nnodes):
        vx = vel[:, i, 0]
        vy = vel[:, i, 1]
        vz = vel[:, i, 2]
        abs_power_energy += masses[i] * (
            np.abs(ag_x * vx) +
            np.abs(ag_y * vy) +
            np.abs(ag_z * vz)
        )
    E_in = np.trapz(abs_power_energy, t)

    # EDP 3
    abs_power = np.zeros(nt)
    for i in range(nnodes):
        vx = vel[:, i, 0]
        vy = vel[:, i, 1]
        vz = vel[:, i, 2]
        abs_power += masses[i] * (
            np.abs(ag_x * vx) +
            np.abs(ag_y * vy) +
            np.abs(ag_z * vz)
        )
    cum_weight = np.cumsum(abs_power * dt)
    cum_weight_norm = cum_weight / cum_weight[-1]
    idx_5 = np.searchsorted(cum_weight_norm, 0.05)
    idx_95 = np.searchsorted(cum_weight_norm, 0.95)
    if idx_5 == idx_95:
        idx_5, idx_95 = 0, len(t) - 1
    D_w_peak = np.max(disp_3d_top[idx_5:idx_95+1])
    t5, t95 = t[idx_5], t[idx_95]

    # EDP 4
    D_3D_max = np.max(disp_3d_top)
    idx_max_3d = np.argmax(disp_3d_top)
    t_max_3d = t[idx_max_3d]

    # EDP 5
    int_vz2 = 0.0
    int_vh2 = 0.0
    for i in range(nnodes):
        vz = vel[:, i, 2]
        vx = vel[:, i, 0]
        vy = vel[:, i, 1]
        int_vz2 += masses[i] * np.trapz(vz**2, t)
        int_vh2 += masses[i] * np.trapz(vx**2 + vy**2, t)
    R_VH = int_vz2 / (int_vh2 + EPS)

    max_uz = np.max(np.abs(uz_top))
    disp_h_top = np.sqrt(ux_top**2 + uy_top**2)
    max_h = np.max(disp_h_top)
    idx_max_uz = np.argmax(np.abs(uz_top))
    t_max_uz = t[idx_max_uz]

    # 返回结果
    return {
        "Case": case_id,
        "Folder": os.path.basename(folder_path),
        "network": event_info.get("network", ""),
        "station": event_info.get("station", ""),
        "event_time": event_info.get("time", ""),
        "magnitude": event_info.get("magnitude", np.nan),
        "D_cum_3D (m·s)": D_cum_3D,
        "E_in (J)": E_in,
        "D_w_peak (m)": D_w_peak,
        "D_3D_max (m)": D_3D_max,
        "R_VH (-)": R_VH,
        "t5 (s)": t5,
        "t95 (s)": t95,
        "t@D3Dmax (s)": t_max_3d,
        "t@maxUz (s)": t_max_uz,
        "max|Uz| (m)": max_uz,
        "max|Uh| (m)": max_h,
        "总质量(动力) (kg)": total_mass,
        "梁质量 (kg)": beam_mass_from_info,
        "附加质量 (kg)": M_add_from_info,
        "步长 dt (s)": dt,
    }

# ============================================================
# 主程序
# ============================================================
def find_folders():
    folders = []
    for f in glob.glob("outputs_mesh_*"):
        if os.path.isdir(f) and re.match(r"^outputs_mesh_\d+$", os.path.basename(f)):
            folders.append(f)
    folders.sort(key=lambda x: int(re.search(r'\d+', x).group()))
    return folders

def extract_case_id(folder_name):
    match = re.search(r'outputs_mesh_(\d+)', folder_name)
    return f"mesh_{match.group(1)}" if match else folder_name

if __name__ == "__main__":
    print("=" * 70)
    print("  OpenSees EDP 后处理 (最终论文版 v7 - 从 model_info.txt 读取事件)")
    print("=" * 70)

    folders = find_folders()
    if not folders:
        print("❌ 未找到任何 outputs_mesh_* 文件夹！")
        sys.exit(1)

    print(f"找到 {len(folders)} 个文件夹：")
    for f in folders:
        print(f"  {f}")

    # 从第一个文件夹读取 model_info 获取事件信息（假设所有文件夹共享同一事件）
    first_folder = folders[0]
    try:
        model_info = read_model_info(first_folder)
        event_info = {
            "network": model_info.get("EventNetwork", ""),
            "station": model_info.get("EventStation", ""),
            "time": model_info.get("EventTime", ""),
            "magnitude": float(model_info.get("EventMagnitude", np.nan))
        }
        print(f"\n事件信息: {event_info['network']}.{event_info['station']}  {event_info['time']}  M{event_info['magnitude']}")
    except:
        print("警告: 无法读取事件信息，将设为空")
        event_info = {"network": "", "station": "", "time": "", "magnitude": np.nan}

    results = []
    for folder in folders:
        try:
            case_id = extract_case_id(os.path.basename(folder))
            print(f"\n处理 {folder} (Case={case_id}) ...")
            res = compute_edps(folder, case_id, event_info)
            results.append(res)
            print(f"\n  D_cum_3D = {res['D_cum_3D (m·s)']:.6e} m·s")
            print(f"  E_in     = {res['E_in (J)']:.6e} J")
            print(f"  D_w_peak = {res['D_w_peak (m)']:.6e} m")
            print(f"  D_3D_max = {res['D_3D_max (m)']:.6e} m")
            print(f"  R_VH     = {res['R_VH (-)']:.6e}")
        except Exception as e:
            print(f"  ❌ 处理失败: {e}")

    if not results:
        sys.exit(1)

    df = pd.DataFrame(results)
    csv_path = "mesh_convergence_EDP_final_v7.csv"
    df.to_csv(csv_path, index=False, float_format="%.6e")

    txt_path = "mesh_convergence_EDP_final_v7.txt"
    with open(txt_path, "w") as f:
        f.write("Mesh Convergence EDP Results (Final v7 - from model_info)\n")
        f.write("=" * 70 + "\n")
        f.write(df.to_string(index=False))
        f.write("\n" + "=" * 70 + "\n")
        f.write("【事件信息】\n")
        f.write(f"  Network: {event_info['network']}\n")
        f.write(f"  Station: {event_info['station']}\n")
        f.write(f"  Time:    {event_info['time']}\n")
        f.write(f"  Magnitude: {event_info['magnitude']}\n")
        f.write("\n【计算说明】\n")
        f.write("1. 地震波为 m/s²，不缩放\n")
        f.write("2. 质量分配：内部节点 beam_mass/numEle，顶部节点 M_ADD + 0.5*beam_mass/numEle\n")
        f.write("3. E_in 为梯形累积绝对输入能量\n")
        f.write("4. D_w_peak 窗口基于各节点绝对功率之和（5%-95%）\n")
        f.write("5. R_VH 采用质量加权速度平方积分\n")
        f.write(f"6. 顶部附加质量 M_ADD = {M_ADD} kg\n")

    print(f"\n✅ 结果已保存至: {csv_path} 和 {txt_path}")
