# ============================================================
# 用户参数
# ============================================================
ACCEL_IN_G = False
EPS = 1.0e-12
M_ADD = 3000.0

F_H = 4.885353e+02
F_V = 8.001221e+01

D = 0.35
t_wall = 0.035
L = 4.5
rho = 2700.0
A_cross = np.pi * ((D/2)**2 - (D/2 - t_wall)**2)
BEAM_MASS = rho * A_cross * L

ACCEL_FILES = {
    'X': "MH1.txt",
    'Y': "MH2.txt",
    'Z': "combine.txt"
}

BASE_WINDOW = 60.0  # 基准窗口（秒）

# ============================================================
# 自动下载
# ============================================================
def ensure_seismic_files(event_index):
    base_dir = f"events/event_{event_index}"
    missing = []
    for f in ACCEL_FILES.values():
        if not os.path.exists(os.path.join(base_dir, f)):
            missing.append(f)
    if missing:
        print(f"地震动文件缺失: {missing}，正在调用 fetch.txt 下载事件 {event_index} ...")
        try:
            subprocess.check_call([sys.executable, "fetch.txt", "--index", str(event_index)])
        except subprocess.CalledProcessError as e:
            raise RuntimeError(f"下载失败: {e}")
        except FileNotFoundError:
            raise RuntimeError("未找到 fetch.txt，请确保该脚本位于当前目录。")
        print("下载完成。")
    else:
        print("所有地震动文件已存在。")

# ============================================================
# 读取函数
# ============================================================
def read_single_node(filepath):
    data = np.loadtxt(filepath)
    if data.ndim == 1:
        data = data.reshape(1, -1)
    t = data[:, 0]
    u = data[:, 1:]  # 6 DOF
    return t, u

def read_accel_file(fpath, dt_ref=0.01):
    data = np.loadtxt(fpath)
    if data.ndim == 1:
        data = data.reshape(-1, 1)
    if data.shape[1] == 1:
        n = data.shape[0]
        t = np.arange(n) * dt_ref
        accel = data[:, 0]
    else:
        t = data[:, 0]
        accel = data[:, 1]
    if ACCEL_IN_G:
        accel = accel * 9.80665
    return t, accel

# ============================================================
# EDP 计算核心（窗口限制）
# ============================================================
def compute_edps(folder_path, case_id, event_info):
    disp_path = os.path.join(folder_path, "disp_all.out")
    vel_path  = os.path.join(folder_path, "vel_all.out")

    t, disp_total = read_single_node(disp_path)
    t_vel, vel_total = read_single_node(vel_path)
    if not np.allclose(t, t_vel):
        raise ValueError("位移与速度时间轴不一致！")

    dt = np.mean(np.diff(t))
    nt = len(t)
    print(f"   读取: {nt} 步, dt = {dt:.6f} s")

    # ---------- 动态位移 ----------
    n_base = int(BASE_WINDOW / dt)
    if n_base > nt:
        n_base = int(0.1 * nt)
    u_base = np.mean(disp_total[:n_base, :], axis=0)
    disp_dyn = disp_total - u_base
    vel_dyn = np.gradient(disp_dyn, dt, axis=0)

    print(f"   基准位移 (前 {n_base*dt:.1f}s 平均): Uz = {u_base[2]:.6e} m")

    # ---------- 读取加速度并插值 ----------
    base_dir = f"events/event_{event_info['index']}"
    t_x, ax = read_accel_file(os.path.join(base_dir, ACCEL_FILES['X']), dt)
    t_y, ay = read_accel_file(os.path.join(base_dir, ACCEL_FILES['Y']), dt)
    t_z, az = read_accel_file(os.path.join(base_dir, ACCEL_FILES['Z']), dt)

    ax = ax / F_H
    ay = ay / F_H
    az = az / F_V

    ag_x = np.interp(t, t_x, ax)
    ag_y = np.interp(t, t_y, ay)
    ag_z = np.interp(t, t_z, az)

    # ---------- 提取平动分量 ----------
    ux = disp_dyn[:, 0]
    uy = disp_dyn[:, 1]
    uz = disp_dyn[:, 2]
    vx = vel_dyn[:, 0]
    vy = vel_dyn[:, 1]
    vz = vel_dyn[:, 2]

    uh = np.sqrt(ux**2 + uy**2)
    u3d = np.sqrt(ux**2 + uy**2 + uz**2)

    # ---------- 计算绝对功率（用于确定显著窗口） ----------
    # 使用顶部节点质量
    mass_top = M_ADD + 0.5 * BEAM_MASS / 10   # numEle = 10
    abs_power = mass_top * (
        np.abs(ag_x * vx) +
        np.abs(ag_y * vy) +
        np.abs(ag_z * vz)
    )

    cum_power = np.cumsum(abs_power * dt)
    cum_power_norm = cum_power / cum_power[-1]
    idx_5 = np.searchsorted(cum_power_norm, 0.05)
    idx_95 = np.searchsorted(cum_power_norm, 0.95)
    if idx_5 == idx_95:
        idx_5, idx_95 = 0, nt - 1

    # ---------- EDP 1: D_cum_3D (窗口内累积) ----------
    D_cum_3D = np.trapz(u3d[idx_5:idx_95+1], t[idx_5:idx_95+1])

    # ---------- EDP 2: E_in (全时程，已随输入变化) ----------
    E_in = np.trapz(abs_power, t)

    # ---------- EDP 3: D_w_peak (窗口内水平位移峰值) ----------
    D_w_peak = np.max(uh[idx_5:idx_95+1])

    # ---------- EDP 4: D_3D_max (窗口内三维位移峰值) ----------
    D_3D_max = np.max(u3d[idx_5:idx_95+1])
    idx_max_3d = idx_5 + np.argmax(u3d[idx_5:idx_95+1])
    t_max_3d = t[idx_max_3d]

    # ---------- EDP 5: R_VH (窗口内动能比) ----------
    int_vz2 = np.trapz(vz[idx_5:idx_95+1]**2, t[idx_5:idx_95+1])
    int_vh2 = np.trapz(vx[idx_5:idx_95+1]**2 + vy[idx_5:idx_95+1]**2, t[idx_5:idx_95+1])
    R_VH = int_vz2 / (int_vh2 + EPS)

    # ---------- 辅助输出 ----------
    max_uz_dyn = np.max(np.abs(uz[idx_5:idx_95+1]))
    max_uh_dyn = np.max(uh[idx_5:idx_95+1])
    idx_max_uz = idx_5 + np.argmax(np.abs(uz[idx_5:idx_95+1]))
    t_max_uz = t[idx_max_uz]
    t5, t95 = t[idx_5], t[idx_95]

    print(f"   窗口: t5={t5:.2f}s, t95={t95:.2f}s")
    print(f"   窗口内 D3Dmax = {D_3D_max:.6e} m")

    return {
        "Case": case_id,
        "Folder": os.path.basename(folder_path),
        "event_index": event_info.get("index", None),
        "network": event_info.get("network", ""),
        "station": event_info.get("station", ""),
        "event_time": event_info.get("time", ""),
        "magnitude": event_info.get("magnitude", np.nan),
        "D_cum_3D (m·s)": D_cum_3D,
        "E_in (J)": E_in,
        "D_w_peak (m)": D_w_peak,
        "D_3D_max (m)": D_3D_max,
        "R_VH (-)": R_VH,
        "t5 (s)": t5,
        "t95 (s)": t95,
        "t@D3Dmax (s)": t_max_3d,
        "t@maxUz (s)": t_max_uz,
        "max|Uz_dyn| (m)": max_uz_dyn,
        "max|Uh_dyn| (m)": max_uh_dyn,
        "基准Uz (m)": u_base[2],
        "窗口内步数": idx_95 - idx_5 + 1,
        "总质量(动力) (kg)": mass_top,
        "梁质量 (kg)": BEAM_MASS,
        "附加质量 (kg)": M_ADD,
        "步长 dt (s)": dt,
    }

# 主程序
# ============================================================
def find_folders(event_index):
    pattern = f"outputs/event_{event_index}/mesh_*"
    folders = glob.glob(pattern)
    folders = [f for f in folders if os.path.isdir(f)]
    folders.sort(key=lambda x: int(re.search(r'mesh_(\d+)', x).group(1)))
    return folders

def extract_case_id(folder_name):
    match = re.search(r'mesh_(\d+)', folder_name)
    if match:
        return f"mesh_{match.group(1)}"
    else:
        return os.path.basename(folder_name)
