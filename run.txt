def find_event_index(network, station, time_str):
    for idx, ev in enumerate(EVENTS, start=1):
        if ev["network"] == network and ev["station"] == station and ev["time"] == time_str:
            return idx
    raise ValueError(f"事件 {network}.{station} at {time_str} 未在列表中")

def ensure_waveforms(event_index, max_retries=3, delay=5):
    """检查波形文件是否存在，若不存在则调用 fetch.txt 下载，最多重试 max_retries 次"""
    base_dir = f"events/event_{event_index}"
    required_files = ["MH1.txt", "MH2.txt", "combine.txt"]
    for attempt in range(1, max_retries + 1):
        missing = []
        for fname in required_files:
            if not os.path.exists(os.path.join(base_dir, fname)):
                missing.append(fname)
        if not missing:
            print(f"所有波形文件已存在: {base_dir}")
            return True
        print(f"波形文件缺失: {missing}，正在尝试下载 (尝试 {attempt}/{max_retries})...")
        try:
            # 调用 fetch.txt 下载，设置超时 120 秒
            subprocess.run(
                [sys.executable, "fetch.txt", "--index", str(event_index)],
                check=True,
                timeout=120
            )
            print("下载完成。")
            # 下载后立即检查文件是否存在
            all_exist = True
            for fname in required_files:
                if not os.path.exists(os.path.join(base_dir, fname)):
                    all_exist = False
                    break
            if all_exist:
                return True
            else:
                print(f"下载后仍有文件缺失，尝试重试...")
        except subprocess.TimeoutExpired:
            print(f"下载超时（120秒），尝试重试...")
        except subprocess.CalledProcessError as e:
            print(f"下载失败 (返回码 {e.returncode})，尝试重试...")
        except FileNotFoundError:
            print("错误：未找到 fetch.txt，请确保该脚本位于当前目录。")
            return False
        if attempt < max_retries:
            print(f"等待 {delay} 秒后重试...")
            time.sleep(delay)
    print(f"在 {max_retries} 次尝试后仍无法获取波形文件。")
    return False

def run_analysis(event_dict, opensees_path="OpenSees", tcl_script="模型.txt", post_script="后处理.txt"):
    try:
        idx = find_event_index(event_dict["network"], event_dict["station"], event_dict["time"])
    except ValueError as e:
        print(f"错误: {e}")
        return False
    print(f"找到事件索引: {idx} ({event_dict['network']}.{event_dict['station']})")

    # 确保波形文件存在
    if not ensure_waveforms(idx):
        print("无法获取波形文件，终止分析。")
        return False

    # 设置环境变量（并打印确认）
    os.environ["SEISMIC_EVENT_INDEX"] = str(idx)
    print(f"环境变量 SEISMIC_EVENT_INDEX = {os.environ['SEISMIC_EVENT_INDEX']}")

    # 运行 OpenSees
    print("\n--- 启动 OpenSees 分析 ---")
    cmd = [opensees_path, tcl_script]
    try:
        # 显式传递当前环境
        result = subprocess.run(cmd, env=os.environ, timeout=3600)  # 1小时超时
    except subprocess.TimeoutExpired:
        print("OpenSees 运行超时（1小时），终止。")
        return False
    except FileNotFoundError:
        print(f"错误：未找到 OpenSees 可执行文件 '{opensees_path}'，请检查路径。")
        return False

    if result.returncode != 0:
        print(f"OpenSees 运行失败，返回码 {result.returncode}")
        return False

    print("OpenSees 分析成功完成。")

    # 运行后处理
    print("\n--- 启动后处理 ---")
    try:
        result_post = subprocess.run([sys.executable, post_script], timeout=300)
    except subprocess.TimeoutExpired:
        print("后处理超时（5分钟），终止。")
        return False
    except FileNotFoundError:
        print(f"错误：未找到后处理脚本 '{post_script}'。")
        return False

    if result_post.returncode != 0:
        print(f"后处理运行失败，返回码 {result_post.returncode}")
        return False

    print("\n✅ 全部完成！")
    return True

    parser.add_argument("--post", default="后处理.txt", help="后处理脚本文件名")
    args = parser.parse_args()

    event_dict = {
        "network": args.network,
        "station": args.station,
        "time": args.time,
        "magnitude": args.magnitude
    }
    success = run_analysis(event_dict, args.opensees, args.tcl, args.post)
    sys.exit(0 if success else 1)
