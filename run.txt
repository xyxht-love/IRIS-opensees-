def find_event_index(event_dict):
    """在 EVENTS 列表中查找匹配的事件索引（基于 network, station, time）"""
    for i, ev in enumerate(EVENTS, start=1):
        if (ev["network"] == event_dict["network"] and
            ev["station"] == event_dict["station"] and
            ev["time"] == event_dict["time"]):
            return i
    raise ValueError(f"未找到匹配的事件: {event_dict}")

def run_for_event(event_dict, opensees_cmd="OpenSees", tcl_script="model_final_science.tcl",
                  postprocess_script="openedp1.py"):
    """
    运行完整分析流程：
    1. 根据 event_dict 查找事件索引。
    2. 设置环境变量 SEISMIC_EVENT_INDEX。
    3. 调用 OpenSees 执行 Tcl 脚本（自动下载波形并分析）。
    4. 调用后处理脚本（自动使用同一事件索引并输出 EDP）。
    """
    # 查找索引
    idx = find_event_index(event_dict)
    print(f"使用事件索引: {idx} -> {event_dict['network']}.{event_dict['station']} ({event_dict['time']})")

    # 设置环境变量（子进程继承）
    env = os.environ.copy()
    env["SEISMIC_EVENT_INDEX"] = str(idx)

    # 步骤1：运行 OpenSees
    print("\n" + "="*70)
    print("步骤1：运行 OpenSees 动力分析")
    print("="*70)
    cmd1 = [opensees_cmd, tcl_script]
    print(f"执行: {' '.join(cmd1)}")
    result1 = subprocess.run(cmd1, env=env, capture_output=False)  # 让输出实时显示
    if result1.returncode != 0:
        raise RuntimeError(f"OpenSees 分析失败，返回码 {result1.returncode}")

    # 步骤2：运行后处理
    print("\n" + "="*70)
    print("步骤2：运行 EDP 后处理")
    print("="*70)
    # 后处理脚本也通过环境变量传递事件索引，或者通过命令行参数
    # 由于 openedp1.py 支持 --event-index，我们优先使用命令行参数，更明确
    cmd2 = [sys.executable, postprocess_script, "--event-index", str(idx)]
    print(f"执行: {' '.join(cmd2)}")
    result2 = subprocess.run(cmd2, env=env, capture_output=False)
    if result2.returncode != 0:
        raise RuntimeError(f"后处理失败，返回码 {result2.returncode}")

    print("\n" + "="*70)
    print("✅ 全部完成！")
    print("="*70)

def main():
    parser = argparse.ArgumentParser(description="根据指定事件运行 OpenSees 分析 + EDP 后处理")
    parser.add_argument("--network", required=True, help="台网")
    parser.add_argument("--station", required=True, help="台站")
    parser.add_argument("--time", required=True, help="事件时间 (ISO 格式)")
    parser.add_argument("--magnitude", type=float, required=True, help="震级")
    parser.add_argument("--opensees", default="OpenSees", help="OpenSees 可执行文件路径")
    parser.add_argument("--tcl", default="model_final_science.tcl", help="Tcl 脚本文件名")
    parser.add_argument("--post", default="openedp1.py", help="后处理脚本文件名")
    args = parser.parse_args()

    event_dict = {
        "network": args.network,
        "station": args.station,
        "time": args.time,
        "magnitude": args.magnitude
    }
    run_for_event(event_dict, opensees_cmd=args.opensees, tcl_script=args.tcl,
                  postprocess_script=args.post)

if __name__ == "__main__":
    main()
