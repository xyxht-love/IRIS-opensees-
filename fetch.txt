# ----------------- 工具函数 -----------------
CHANNEL_SCHEMES = [["BH1", "BH2", "BHZ"], ["BHN", "BHE", "BHZ"]]

def check_components(st):
    comp = {"H1": None, "H2": None, "V": None}
    for tr in st:
        cha = tr.stats.channel
        last = cha[-1]
        if last in ("1", "N"):
            comp["H1"] = tr
        elif last in ("2", "E"):
            comp["H2"] = tr
        elif last == "Z":
            comp["V"] = tr
        if cha.startswith("BH"):
            if cha.endswith(("1", "N")):
                comp["H1"] = tr
            elif cha.endswith(("2", "E")):
                comp["H2"] = tr
            elif cha.endswith("Z"):
                comp["V"] = tr
    return comp

def get_event_waveforms(event, before=60, after=3600):
    client = Client(FDSN_CLIENT)
    starttime = event.time - before
    endtime = event.time + after
    for loc in ["00", "10", ""]:
        for scheme in CHANNEL_SCHEMES:
            try:
                st = client.get_waveforms(
                    network=event.network,
                    station=event.station,
                    location=loc,
                    channel=",".join(scheme),
                    starttime=starttime,
                    endtime=endtime,
                    attach_response=False
                )
                if len(st) >= 3:
                    comp = check_components(st)
                    if all(v is not None for v in comp.values()):
                        st.merge(fill_value="interpolate")
                        return st
            except:
                continue
    raise RuntimeError(f"无法获取 {event.network}.{event.station} 三分量波形")

def get_inventory(network, station, starttime, endtime):
    cache_file = os.path.join(XML_CACHE_DIR, f"{network}_{station}.xml")
    if os.path.exists(cache_file):
        try:
            inv = read_inventory(cache_file)
            return inv
        except:
            pass
    client = Client(FDSN_CLIENT)
    inv = client.get_stations(
        network=network, station=station, channel="BH?",
        location="*", starttime=starttime, endtime=endtime,
        level="response"
    )
    inv.write(cache_file, format="STATIONXML")
    return inv

def preprocess_stream(st, inv):
    st_acc = st.copy()
    for tr in st_acc:
        resp = inv.select(network=tr.stats.network, station=tr.stats.station,
                          location=tr.stats.location, channel=tr.stats.channel)
        tr.remove_response(inventory=resp, output="ACC",
                           pre_filt=PRE_FILTER, water_level=WATER_LEVEL)
        tr.detrend("demean")
        tr.detrend("linear")
        tr.taper(max_percentage=TAPER_PERCENT, type="cosine")
        tr.filter("bandpass", freqmin=FILTER_FREQMIN, freqmax=FILTER_FREQMAX,
                  corners=4, zerophase=True)
        tr.data = tr.data.astype(np.float64)
    return st_acc

# ----------------- 主函数 -----------------
def fetch_event(index):
    if index < 1 or index > len(EVENTS):
        raise ValueError(f"事件序号应在 1 ~ {len(EVENTS)} 之间")
    info = EVENTS[index-1]
    event = type('Event', (), {})()
    event.network = info["network"]
    event.station = info["station"]
    event.time = UTCDateTime(info["time"])
    event.magnitude = info.get("magnitude")

    print(f"正在下载事件 {index}: {event.network}.{event.station} ({event.time})")
    st = get_event_waveforms(event)
    inv = get_inventory(event.network, event.station,
                        event.time - 60, event.time + 3600)
    st_acc = preprocess_stream(st, inv)
    comp = check_components(st_acc)
    if None in comp.values():
        raise RuntimeError("预处理后缺少三分量")

    # 确定输出目录
    out_dir = f"events/event_{index}"
    os.makedirs(out_dir, exist_ok=True)

    # 输出文件到子目录
    dt = comp["H1"].stats.delta
    for key, fname in [("H1", "MH1.txt"), ("H2", "MH2.txt"), ("V", "combine.txt")]:
        tr = comp[key]
        data = tr.data
        n = len(data)
        times = np.arange(n) * dt
        out = np.column_stack((times, data))
        out_path = os.path.join(out_dir, fname)
        np.savetxt(out_path, out, fmt="%.8e")
        print(f"已输出 {out_path} ({n} 点, dt={dt:.6f}s)")

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="从 IRIS 获取地震动数据")
    parser.add_argument("--index", type=int, required=True, help="事件序号 (1~40)")
    args = parser.parse_args()
    fetch_event(args.index)
